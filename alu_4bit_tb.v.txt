`timescale 1ns/1ps

module alu_4bit_tb;

    reg  [3:0] A;
    reg  [3:0] B;
    reg  [2:0] sel;
    wire [3:0] result;
    wire       carry_out;
    wire       zero;

    // Instantiate Unit Under Test (UUT)
    alu_4bit uut (
        .A(A),
        .B(B),
        .sel(sel),
        .result(result),
        .carry_out(carry_out),
        .zero(zero)
    );

    initial begin
        $display("Time | sel |   A    |   B    | result | carry | zero");
        $display("-----------------------------------------------------");
        $monitor("%4t | %b | %b (%2d) | %b (%2d) |  %b (%2d) |   %b   |   %b",
                 $time, sel, A, A, B, B, result, result, carry_out, zero);

        // Test 1: Addition with carry (9 + 8 = 17 -> 4'b0001, carry=1)
        A = 4'd9; B = 4'd8; sel = 3'b000; #10;

        // Test 2: Subtraction resulting in zero (6 - 6 = 0 -> zero=1)
        A = 4'd6; B = 4'd6; sel = 3'b001; #10;

        // Test 3: Bitwise AND
        A = 4'b1100; B = 4'b1010; sel = 3'b010; #10;

        // Test 4: Bitwise OR
        A = 4'b1100; B = 4'b1010; sel = 3'b011; #10;

        // Test 5: Bitwise XOR
        A = 4'b1100; B = 4'b1010; sel = 3'b100; #10;

        // Test 6: Bitwise NOT of A
        A = 4'b1111; B = 4'b0000; sel = 3'b101; #10;

        // Test 7: Shift Left by 1
        A = 4'b1011; B = 4'b0000; sel = 3'b110; #10;

        // Test 8: Shift Right by 1
        A = 4'b1011; B = 4'b0000; sel = 3'b111; #10;

        $display("-----------------------------------------------------");
        $display("Simulation completed.");
        $finish;
    end

endmodule
